//@version=6
strategy("M-O-8 (Optimized) STRATEGY", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=10, initial_capital=100000, commission_type=strategy.commission.percent, commission_value=0.04, pyramiding=1)

// ─────────────────────────────────────────────────────────────
// Strategy Settings (new)
// ─────────────────────────────────────────────────────────────
tpPoints   = input.int(150, "Take-Profit (points)", group="Strategy Settings")
slPoints   = input.int(80,  "Stop-Loss (points)",   group="Strategy Settings")
tpPercent  = input.float(1.2, "Take-Profit (% of entry price)", step=0.1, group="Strategy Settings")
slPercent  = input.float(0.8, "Stop-Loss (% of entry price)", step=0.1, group="Strategy Settings")
useATR     = input.bool(false, "Use ATR for TP/SL", group="Strategy Settings")
atrLen     = input.int(14, "ATR Length", group="Strategy Settings")
atrMultTP  = input.float(2.0, "ATR × for TP", step=0.1, group="Strategy Settings")
atrMultSL  = input.float(1.0, "ATR × for SL", step=0.1, group="Strategy Settings")

// Strategy Modes – choose which signals to trade
useBasic   = input.bool(true,  "Use Basic Composite Crossover", group="Strategy Modes")
useVShape1  = input.bool(true,  "Use V-Shape Reversal",         group="Strategy Modes")
useMTF     = input.bool(false, "Require MTF Confluence (30m+1h+4h)", group="Strategy Modes")
useCompositeOnly = input.bool(false, "Ignore V-Shape & MTF – trade Composite only", group="Strategy Modes")

// ─────────────────────────────────────────────────────────────
// Helper: normalize to 0–100 over a rolling window
// ─────────────────────────────────────────────────────────────
norm(src, length) =>
    hi = ta.highest(src, length)
    lo = ta.lowest(src, length)
    hi != lo ? (src - lo) / (hi - lo) * 100 : 50

// ─────────────────────────────────────────────────────────────
// Inputs
// ─────────────────────────────────────────────────────────────
rsiLen        = input.int(7,  "RSI Length")
stoLen        = input.int(7,  "Stoch %K Length")
demLen        = input.int(14,  "DeMarker Length")
wrLen         = input.int(80,  "Williams %R Length")
cciLen        = input.int(35,  "CCI Length")
rviLen        = input.int(20,  "RVI Length")
fiLen         = input.int(50,  "Force Index Length")
stdLen        = input.int(30,  "Std Dev Length")

lookbackBars  = input.int(20,  "Lookback Bars", minval=1)
useLookback   = input.bool(true, "Use Lookback Filter (reversal gating)")

showArrows    = input.bool(true, "Show Arrows on Price")
showComposite = input.bool(true, "Show Composite")

osLevel       = input.int(27, "Oversold Level", minval=0, maxval=100)
obLevel       = input.int(73, "Overbought Level", minval=0, maxval=100)

// V-Shape (major reversal) settings
useVShape     = input.bool(true, "Use V-Shape Major Reversal Signals")
pivotLR       = input.int(3, "V Pivot Strength (L/R bars)", minval=1, maxval=10)
requireDepth  = input.float(0.0, "Min Depth past OS/OB (0 = off)", step=0.1)

pvtNormLength = input.int(100, "PVT Normalization Length", minval=1)

// Show Individual Oscillators
showParts = input.bool(true, "Show Individual Oscillators")

// MTF Status Toggles
show30m = input.bool(true, "Show 30m Status")
show1h  = input.bool(true, "Show 1h Status")
show4h  = input.bool(true, "Show 4h Status")
showMTFMarks = input.bool(true, "Show MTF Marks (pane)")

// ─────────────────────────────────────────────────────────────
// Core calc: PVT
// ─────────────────────────────────────────────────────────────
src = close
pvt = ta.cum(ta.change(src) / src[1] * volume)

// ─────────────────────────────────────────────────────────────
// Compute each normalized oscillator (0–100 scale)
// ─────────────────────────────────────────────────────────────
rsiVal  = ta.rsi(close, rsiLen)

// NOTE: ta.stoch is already 0–100 in Pine.
// Keep your intended behavior (normalize again) but remove the *100 multiplier.
stoBase = ta.stoch(close, high, low, stoLen)
stoVal  = norm(stoBase, stoLen * 3)

up      = math.max(high - high[1], 0)
dn      = math.max(low[1] - low,   0)
demVal  = math.sum(up, demLen) / (math.sum(up, demLen) + math.sum(dn, demLen)) * 100

wrVal   = ta.wpr(wrLen) + 100

cciVal  = norm(ta.cci(close, cciLen), cciLen * 3)

rviRaw  = ta.sma(close - open, rviLen) / ta.sma(high - low, rviLen)
rviVal  = norm(rviRaw, rviLen * 3)

fiRaw   = ta.ema((close - close[1]) * volume, fiLen)
fiVal   = norm(fiRaw, fiLen * 5)

sdRaw   = ta.stdev(close, stdLen)
sdVal   = norm(sdRaw, stdLen * 5)

pvtNorm = norm(pvt, pvtNormLength)

// ─────────────────────────────────────────────────────────────
// Composite (9 components)
// ─────────────────────────────────────────────────────────────
composite = (rsiVal + stoVal + demVal + wrVal + cciVal + rviVal + fiVal + sdVal + pvtNorm) / 9.0

// MTF Consensus Flags (properly calculated on their timeframes)
exitOB_30  = show30m ? request.security(syminfo.tickerid, "30",  ta.crossunder(composite, obLevel), lookahead=barmerge.lookahead_off) : false
exitOS_30  = show30m ? request.security(syminfo.tickerid, "30",  ta.crossover(composite,  osLevel), lookahead=barmerge.lookahead_off) : false

exitOB_60  = show1h  ? request.security(syminfo.tickerid, "60",  ta.crossunder(composite, obLevel), lookahead=barmerge.lookahead_off) : false
exitOS_60  = show1h  ? request.security(syminfo.tickerid, "60",  ta.crossover(composite,  osLevel), lookahead=barmerge.lookahead_off) : false

exitOB_240 = show4h  ? request.security(syminfo.tickerid, "240", ta.crossunder(composite, obLevel), lookahead=barmerge.lookahead_off) : false
exitOS_240 = show4h  ? request.security(syminfo.tickerid, "240", ta.crossover(composite,  osLevel), lookahead=barmerge.lookahead_off) : false

// Consensus flags used by the strategy
allBear = exitOB_30 and exitOB_60 and exitOB_240
allBull = exitOS_30 and exitOS_60 and exitOS_240

plot(showComposite ? composite : na, title="Composite", color=color.white, linewidth=2)

// Levels + fills (single, clean set)
h50 = hline(50,      "Mid", color=color.new(color.gray, 0))
hOB = hline(obLevel, "OB",  color=color.new(color.red,  0), linestyle=hline.style_dashed)
hOS = hline(osLevel, "OS",  color=color.new(color.green,0), linestyle=hline.style_dashed)

fill(hOB, h50, color=color.new(color.red,   92))
fill(hOS, h50, color=color.new(color.green, 92))

// Optional “active zone” shading
bgcolor(composite > obLevel ? color.new(color.red, 92) : composite < osLevel ? color.new(color.green, 92) : na)

// ─────────────────────────────────────────────────────────────
// Signal throttling (one signal per swing)
// ─────────────────────────────────────────────────────────────
var bool allowLong = true
var bool allowShort = true

// Reset permissions when opposite extreme is reached
if composite < osLevel
    allowLong := true

if composite > obLevel
    allowShort := true

// ─────────────────────────────────────────────────────────────
// Reversal gating: require we've visited the zone recently
// ─────────────────────────────────────────────────────────────
everInOS = ta.barssince(composite < osLevel) < lookbackBars
everInOB = ta.barssince(composite > obLevel) < lookbackBars

// ─────────────────────────────────────────
// V-shape reversal logic
// ─────────────────────────────────────────
vLow  = ta.pivotlow(composite, pivotLR, pivotLR)
vHigh = ta.pivothigh(composite, pivotLR, pivotLR)

vLowVal  = composite[pivotLR]
vHighVal = composite[pivotLR]

// pivot must be beyond OS/OB
vBuyPivotOK  = not na(vLow)  and (vLowVal  < (osLevel - requireDepth))
vSellPivotOK = not na(vHigh) and (vHighVal > (obLevel + requireDepth))

// confirmation exit from extreme
exitOS = ta.crossover(composite, osLevel)
exitOB = ta.crossunder(composite, obLevel)

// V-shape signals
rawBuyV  = vBuyPivotOK  and exitOS
rawSellV = vSellPivotOK and exitOB

// fallback basic signals if V-shape disabled
rawBuyBasic  = ta.crossover(composite, osLevel)
rawSellBasic = ta.crossunder(composite, obLevel)

rawBuy  = useVShape ? rawBuyV  : rawBuyBasic
rawSell = useVShape ? rawSellV : rawSellBasic

buySignalRaw  = useLookback ? (rawBuy  and everInOS) : rawBuy
sellSignalRaw = useLookback ? (rawSell and everInOB) : rawSell

// throttle signals per swing
buySignal  = buySignalRaw and allowLong
sellSignal = sellSignalRaw and allowShort

if buySignal
    allowLong := false

if sellSignal
    allowShort := false

plotshape(useVShape and vBuyPivotOK  ? osLevel : na, title="V Low (confirmed)", style=shape.circle, location=location.belowbar, color=color.new(color.blue, 20), size=size.tiny, force_overlay = true)
plotshape(useVShape and vSellPivotOK ? obLevel : na, title="V High (confirmed)", style=shape.circle, location=location.abovebar, color=color.new(color.orange, 20), size=size.tiny, force_overlay = true)

// ───────────────────────────────────────────────────────────── 
// FINAL STRATEGY SIGNALS 
// ───────────────────────────────────────────────────────────── 
buyCondition  = false 
sellCondition = false 

if useCompositeOnly
    // Ignore V-shape and MTF - just use basic composite crossovers
    buyCondition  := buySignalRaw
    sellCondition := sellSignalRaw
else if useMTF
    // Require MTF confluence
    buyCondition  := buySignal and allBull
    sellCondition := sellSignal and allBear
else if useVShape1
// Use V-shape reversal with throttling
    buyCondition  := buySignal
    sellCondition := sellSignal
else if useBasic
    // Basic composite crossover
    buyCondition  := buySignalRaw
    sellCondition := sellSignalRaw
    
// ─────────────────────────────────────────────────────────────
// EXECUTION
// ─────────────────────────────────────────────────────────────
if buyCondition
    strategy.entry("Long", strategy.long)

if sellCondition
    strategy.entry("Short", strategy.short)

// TP / SL (flexible) 
atrVal = ta.atr(atrLen) 

if useATR
    tpDist = atrVal * atrMultTP
    slDist = atrVal * atrMultSL
    strategy.exit("Long Exit",  from_entry="Long",  profit=tpDist / syminfo.mintick, loss=slDist / syminfo.mintick)
    strategy.exit("Short Exit", from_entry="Short", profit=tpDist / syminfo.mintick, loss=slDist / syminfo.mintick)
else
    strategy.exit("Long Exit",  from_entry="Long",  profit=tpPoints, loss=slPoints)
    strategy.exit("Short Exit", from_entry="Short", profit=tpPoints, loss=slPoints)

// ─────────────────────────────────────────────
// EXPORT (for master strategy to read)
// ─────────────────────────────────────────────
plot(buySignal  ? 1 : 0, "KB_MO8_EXPORT_BUY",  display=display.none)
plot(sellSignal ? 1 : 0, "KB_MO8_EXPORT_SELL", display=display.none)

// ─────────────────────────────────────────────────────────────
// Price arrows (optional)
// ─────────────────────────────────────────────────────────────
plotchar(showArrows and buySignal,  title="Buy",  char="✳️", location=location.belowbar, color=color.green, size=size.tiny, force_overlay=true)
plotchar(showArrows and sellSignal, title="Sell", char="🔆", location=location.abovebar, color=color.red,   size=size.tiny, force_overlay=true)

// ─────────────────────────────────────────────────────────────
// MTF marks in oscillator pane (minimal clutter)
// ─────────────────────────────────────────────────────────────
// 30m
plotshape(showMTFMarks and exitOB_30 ? obLevel : na, title="OB 30m", style=shape.triangledown, location=location.absolute, size=size.tiny, color=color.new(color.blue, 0))
plotshape(showMTFMarks and exitOS_30 ? osLevel : na, title="OS 30m", style=shape.triangleup, location=location.absolute, size=size.tiny, color=color.new(color.blue, 0))

// 1h
plotshape(showMTFMarks and exitOB_60 ? obLevel : na, title="OB 1h", style=shape.circle, location=location.absolute, size=size.tiny, color=color.new(color.white, 55))
plotshape(showMTFMarks and exitOS_60 ? osLevel : na, title="OS 1h", style=shape.circle, location=location.absolute, size=size.tiny, color=color.new(color.white, 55))

// 4h
plotshape(showMTFMarks and exitOB_240 ? obLevel : na, title="OB 4h", style=shape.xcross, location=location.absolute, size=size.tiny, color=color.new(color.red, 55))
plotshape(showMTFMarks and exitOS_240 ? osLevel : na, title="OS 4h", style=shape.xcross, location=location.absolute, size=size.tiny, color=color.new(color.red, 55))

// Individual Indicator Lines
plot(showParts ? rsiVal  : na, color=color.new(color.yellow, 50), title="RSI")
plot(showParts ? stoVal  : na, color=color.new(color.blue, 50), title="Stoch")
plot(showParts ? demVal  : na, color=color.new(color.orange, 50), title="DeMarker")
plot(showParts ? wrVal   : na, color=color.new(color.red, 50), title="Williams")
plot(showParts ? cciVal  : na, color=color.new(color.teal, 50), title="CCI")
plot(showParts ? rviVal  : na, color=color.new(color.purple, 50), title="RVI")
plot(showParts ? fiVal   : na, color=color.new(color.green, 50), title="Force")
plot(showParts ? sdVal   : na, color=color.new(color.gray, 50), title="StdDev")
plot(showParts ? pvtNorm : na, color=color.new(color.white, 50), title="PVT")
