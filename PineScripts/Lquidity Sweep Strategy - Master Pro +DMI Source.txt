tp_l        = f_toTicks(tp_l_input)
sl_l        = f_toTicks(sl_l_input)

trailStep_l = useTrail ? f_toTicks(trailStep_l_input) : na

tp_s        = f_toTicks(tp_s_input)
sl_s        = f_toTicks(sl_s_input)
trailAct_s  = useTrail ? f_toTicks(trailAct_s_input)  : na
trailStep_s = useTrail ? f_toTicks(trailStep_s_input) : na

tfRatio = 60.0 / (timeframe.isminutes ? timeframe.multiplier : 11.0)
localHtfLen_l = math.max(2, math.round(htfLen_l * tfRatio))
localHtfLen_s = math.max(2, math.round(htfLen_s * tfRatio))
localHtfOff_l = math.round(htfOff_l * tfRatio)
localHtfOff_s = math.round(htfOff_s * tfRatio)

// =========================================================================

finalSrc_l    = useDmiAsSource ? diPlusRaw  : maSrc_l
finalSrc_s    = useDmiAsSource ? diMinusRaw : maSrc_s
finalHtfSrc_l = useDmiAsSource ? diPlusRaw  : htfSrc_l
finalHtfSrc_s = useDmiAsSource ? diMinusRaw : htfSrc_s

barsSinceClose = ta.barssince(strategy.position_size[1] != 0 and strategy.position_size == 0)
cooldownPassed = na(barsSinceClose) or (barsSinceClose >= cooldownBars)

// =========================================================================
// MA ENGINE CALCULATIONS
// =========================================================================
maVal_l = switch maType_l
    "EMA"  => ta.ema(finalSrc_l, maLen_l)
    "SMA"  => ta.sma(finalSrc_l, maLen_l)
    "LSMA" => ta.linreg(finalSrc_l, maLen_l, maOff_l)
    =>        ta.linreg(finalSrc_l, maLen_l, maOff_l)

maVal_s = switch maType_s
    "EMA"  => ta.ema(finalSrc_s, maLen_s)
    "SMA"  => ta.sma(finalSrc_s, maLen_s)
    "LSMA" => ta.linreg(finalSrc_s, maLen_s, maOff_s)
    =>        ta.linreg(finalSrc_s, maLen_s, maOff_s)

htfMA_l = switch htfType_l
    "EMA"  => ta.ema(finalHtfSrc_l, localHtfLen_l)
    "SMA"  => ta.sma(finalHtfSrc_l, localHtfLen_l)
    "LSMA" => ta.linreg(finalHtfSrc_l, localHtfLen_l, localHtfOff_l)
    =>        ta.linreg(finalHtfSrc_l, localHtfLen_l, localHtfOff_l)

htfMA_s = switch htfType_s
    "EMA"  => ta.ema(finalHtfSrc_s, localHtfLen_s)
    "SMA"  => ta.sma(finalHtfSrc_s, localHtfLen_s)
    "LSMA" => ta.linreg(finalHtfSrc_s, localHtfLen_s, localHtfOff_s)
    =>        ta.linreg(finalHtfSrc_s, localHtfLen_s, localHtfOff_s)

// =========================================================================
// FLUX CHARTS LOGIC (KDE PERCENTILE CALCULATOR)
// =========================================================================
gaussian(float distance, float bandwidth = 1.0) => 
    1.0 / math.sqrt(2.0 * math.pi) * math.pow(math.e, -0.5 * math.pow(distance / bandwidth, 2.0))

type KDE_Engine
    float mid = na
    float[] X = na
    float[] Y = na
    float[] YSum = na

f_kde(array<float> arr, float bandwidth, int steps) =>
    arrMax = 3.0, arrMin = -3.0, stepCount = (arrMax - arrMin) / steps
    densityRange = array.new<float>(steps)
    for i = 0 to (steps) - 1
        densityRange.set(i, arrMin + i * stepCount)
    xArr = array.new<float>(), yArr = array.new<float>()
    for i = 0 to densityRange.size() - 1
        float temp = 0.0
        for j = 0 to arr.size() - 1
            temp += gaussian(densityRange.get(i) - arr.get(j), bandwidth)
        xArr.push(densityRange.get(i)), yArr.push(temp)
    totalDensity = array.sum(yArr)
    for k = 0 to yArr.size() - 1
        yArr.set(k, yArr.get(k) / (totalDensity > 0 ? totalDensity : 1.0))
    [xArr, yArr]

f_prefixSum(array<float> arr, int l, int r) => arr.get(r) - (l == 0 ? 0.0 : arr.get(l - 1))
f_standardize(float val, array<float> arr) => (val - array.avg(arr)) / (array.stdev(arr) > 0 ? array.stdev(arr) : 1.0)

[Pine_Supertrend, pineDirection] = ta.supertrend(atrMultiplier, atrLength)
bullishBreak = ta.crossover(close, Pine_Supertrend)
bearishBreak = ta.crossunder(close, Pine_Supertrend)

buyVolumeTotal  = volume * (high - low > 0 ? (close - low) / (high - low) : 0.5)
sellVolumeTotal = volume * (high - low > 0 ? (high - close) / (high - low) : 0.5)

emaBuyDenom  = ta.ema(buyVolumeTotal, barLength)
emaSellDenom = ta.ema(sellVolumeTotal, barLength)

buyVolume  = ta.ema(buyVolumeTotal, 10) / (emaBuyDenom > 0 ? emaBuyDenom : 1.0)
sellVolume = ta.ema(sellVolumeTotal, 10) / (emaSellDenom > 0 ? emaSellDenom : 1.0)

var volumeOnBullishBreak = array.new<float>()
var volumeOnBearishBreak = array.new<float>()
var bullishKDE = KDE_Engine.new(na, array.new<float>(), array.new<float>(), array.new<float>())
var bearishKDE = KDE_Engine.new(na, array.new<float>(), array.new<float>(), array.new<float>())

if not na(volume)
    if bullishBreak and not na(buyVolume)
        if volumeOnBullishBreak.size() >= KDEStep                  
            volumeOnBullishBreak.shift()
        volumeOnBullishBreak.push(buyVolume)
        [bX, bY] = f_kde(array.standardize(volumeOnBullishBreak), KDEBandwidth, KDEStep)
        bullishKDE.X := bX, bullishKDE.Y := bY, bullishKDE.YSum.clear()
        temp = 0.0
        for i = 0 to bullishKDE.Y.size() - 1
            temp += bullishKDE.Y.get(i)                            
            bullishKDE.YSum.push(temp)
    if bearishBreak and not na(sellVolume)
        if volumeOnBearishBreak.size() >= KDEStep                  
            volumeOnBearishBreak.shift()
        volumeOnBearishBreak.push(sellVolume)
        [sX, sY] = f_kde(array.standardize(volumeOnBearishBreak), KDEBandwidth, KDEStep)
        bearishKDE.X := sX, bearishKDE.Y := sY, bearishKDE.YSum.clear()
        temp = 0.0
        for i = 0 to bearishKDE.Y.size() - 1
            temp += bearishKDE.Y.get(i)                            
            bearishKDE.YSum.push(temp)

var float volumeProb = 1.0
bull = pineDirection < 0 and not na(volume) 
bear = pineDirection > 0 and not na(volume) 

if bull and volumeOnBullishBreak.size() > 0 and bullishKDE.X.size() > 0
    vol = f_standardize(buyVolume, volumeOnBullishBreak)
    vL = array.binary_search_leftmost(bullishKDE.X, vol), vR = math.min(array.binary_search_rightmost(bullishKDE.X, vol), bullishKDE.X.size() - 1)
    volumeProb := f_prefixSum(bullishKDE.YSum, 0, (math.abs(vol - bullishKDE.X.get(vL)) < math.abs(vol - bullishKDE.X.get(vR))) ? vL : vR)

if bear and volumeOnBearishBreak.size() > 0 and bearishKDE.X.size() > 0
    vol = f_standardize(sellVolume, volumeOnBearishBreak)
    vL = array.binary_search_leftmost(bearishKDE.X, vol), vR = math.min(array.binary_search_rightmost(bearishKDE.X, vol), bearishKDE.X.size() - 1)
    volumeProb := f_prefixSum(bearishKDE.YSum, 0, (math.abs(vol - bearishKDE.X.get(vL)) < math.abs(vol - bearishKDE.X.get(vR))) ? vL : vR)

kdeVolumeFilterPass = (not useKdeFilter) or (volumeProb >= activationThreshold)

// =========================================================================
// SETUP STRUCTURE BRACKETS & VISUAL LINES
// =========================================================================
isLiqCandle = high > high[1] and low < low[1]
barcolor(isLiqCandle ? color.white : na)

var line lineHigh = na, var line lineLow  = na
if isLiqCandle
    lineHigh := line.new(bar_index, high, bar_index, high, color=color.green, width=1)
    lineLow  := line.new(bar_index, low,  bar_index, low,  color=color.red,   width=1)
else if not na(lineHigh)
    line.set_x2(lineHigh, bar_index), line.set_x2(lineLow,  bar_index)

// =========================================================================
// RUNTIME ENTRY CALCULATIONS (WITH ADAPTIVE DMI SCALE FILTER GATES)
// =========================================================================
var float buyLevel  = na 
var float sellLevel = na
barsSinceLiq = ta.barssince(isLiqCandle)

// SMART SCALE COMPARISONS: Price vs Price MA | OR | DMI Line vs DMI MA
longMA_pass   = useDmiAsSource ? (diPlusRaw > maVal_l)   : (close > maVal_l)
longHtf_pass  = useDmiAsSource ? (diPlusRaw > htfMA_l)  : (close > htfMA_l)

shortMA_pass  = useDmiAsSource ? (diMinusRaw > maVal_s)  : (close < maVal_s)
shortHtf_pass = useDmiAsSource ? (diMinusRaw > htfMA_s) : (close < htfMA_s)

// Extra gate to guarantee we don't buy if DI- dominates DI+
dmiTrendGateLong  = useDmiAsSource ? (diPlusRaw > diMinusRaw) : true
dmiTrendGateShort = useDmiAsSource ? (diMinusRaw > diPlusRaw) : true

longCondition  = (not useTF_l or longMA_pass) and (not useHTF_l or longHtf_pass) and dmiTrendGateLong and kdeVolumeFilterPass and cooldownPassed
shortCondition = (not useTF_s or shortMA_pass) and (not useHTF_s or shortHtf_pass) and dmiTrendGateShort and kdeVolumeFilterPass and cooldownPassed

if strategy.position_size == 0
    if isLiqCandle
        buyLevel  := allowLongs   ? high + offsetPrice_l : na
        sellLevel := allowShorts  ? low  - offsetPrice_s : na
        
    if barsSinceLiq > barWindow_l
        buyLevel := na
        strategy.cancel("Long")
        
    if barsSinceLiq > barWindow_s
        sellLevel := na
        strategy.cancel("Short")

if strategy.position_size == 0
    if not na(buyLevel) and longCondition
        strategy.entry("Long", strategy.long, stop=buyLevel)
        
    if not na(sellLevel) and shortCondition
        strategy.entry("Short", strategy.short, stop=sellLevel)
else
    buyLevel := na
    sellLevel := na
    strategy.cancel_all()

// =========================================================================
// PROTECTIVE EXIT BRACKETS
// =========================================================================
if strategy.position_size > 0
    if useTrail
        strategy.exit("Exit Long", "Long", profit=tp_l, loss=sl_l, trail_points=trailAct_l, trail_offset=trailStep_l)
    else
        strategy.exit("Exit Long", "Long", profit=tp_l, loss=sl_l)

if strategy.position_size < 0
    if useTrail
        strategy.exit("Exit Short", "Short", profit=tp_s, loss=sl_s, trail_points=trailAct_s, trail_offset=trailStep_s)
    else
        strategy.exit("Exit Short", "Short", profit=tp_s, loss=sl_s)

// Visual plots
plot(allowLongs and useTF_l and not useDmiAsSource ? maVal_l : na, title="Long Current Price MA", color=color.lime, linewidth=1)
plot(allowShorts and useTF_s and not useDmiAsSource ? maVal_s : na, title="Short Current Price MA", color=color.red, linewidth=1)
