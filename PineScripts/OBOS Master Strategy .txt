// ============================================================================
// SIGNAL STRENGTH LOGIC
// ============================================================================
// Weak = only one engine active
weakLong =
     (rsi_obPressureActive and not st_obPressureActive) or
     (st_obPressureActive and not rsi_obPressureActive)

weakShort =
     (rsi_osPressureActive and not st_osPressureActive) or
     (st_osPressureActive and not rsi_osPressureActive)

// Strong = both engines aligned
strongLong =
     rsi_obPressureActive and st_obPressureActive

strongShort =
     rsi_osPressureActive and st_osPressureActive

// --- UPDATED: TRIGGER LOGIC ---
// This prevents "Weak" signals from firing if we are transitioning DOWN from a "Strong" state.
// It also ensures shapes only fire once when the state begins.

// Only trigger a Weak signal if we were Neutral before (not Strong and not Weak)
triggerWeakLong    = weakLong and not (weakLong[1] or strongLong[1])
triggerWeakShort   = weakShort and not (weakShort[1] or strongShort[1])

// Strong signals stay the same (fire when moving from Weak or Neutral to Strong)

// Entries
if triggerStrongLong and not strongShort
    strategy.entry("Long", strategy.long)

if triggerStrongShort and not strongLong
    strategy.entry("Short", strategy.short)

// --- DYNAMIC EXIT CALCULATIONS ---
// We use 'var' to lock in the price when the trade opens so it doesn't drift
var float slPrice = na
var float tpPrice = na

// When a new position is opened, calculate the SL/TP levels
if strategy.position_size != strategy.position_size[1]
    if strategy.position_size > 0
        slPrice := strategy.position_avg_price * (1 - slPercent / 100)
        tpPrice := strategy.position_avg_price * (1 + tpPercent / 100)
    else if strategy.position_size < 0
        slPrice := strategy.position_avg_price * (1 + slPercent / 100)
        tpPrice := strategy.position_avg_price * (1 - tpPercent / 100)
    else
        slPrice := na
        tpPrice := na

// ============================================================================
// 1H SMA EXECUTION
// ============================================================================
// Strong entries with Trend Filter applied
if triggerStrongLong and not strongShort and canLong
    strategy.entry("Long", strategy.long)

if triggerStrongShort and not strongLong and canShort
    strategy.entry("Short", strategy.short)

strategy.exit("Exit Short", from_entry = "Short", 
    stop  = useSL ? slPrice : na, 
    limit = useTP ? tpPrice : na,
    comment_loss = "SL", comment_profit = "TP")
